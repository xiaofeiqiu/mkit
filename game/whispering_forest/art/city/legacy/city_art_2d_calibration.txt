static func basis(asset: String) -> Transform2D:
	var slopes: Vector2 = BUILDINGS[asset].slopes
	var vertical := 1.0/(slopes.x-slopes.y)
	return Transform2D(Vector2(1,0.5-vertical*slopes.x),Vector2(0,vertical),Vector2.ZERO)

static func source_bounds(asset: String) -> Rect2:
	var projection := basis(asset)
	var points: Array = BUILDINGS[asset].base
	var rect := Rect2(WFIso.unproject(projection*points[0]),Vector2.ZERO)
	for point in points: rect = rect.expand(WFIso.unproject(projection*point))
	return rect

static func scale_for(asset: String,width: float) -> float:
	return width/float(REGIONS[asset].region[2])

static func footprint(asset: String,width: float) -> Vector2:
	return source_bounds(asset).size*scale_for(asset,width)

static func pivot(asset: String) -> Vector2:
	return basis(asset).affine_inverse()*WFIso.project(source_bounds(asset).end)

static func door_offset(asset: String,width: float) -> Vector2:
	return (WFIso.unproject(basis(asset)*BUILDINGS[asset].door)-source_bounds(asset).end)*scale_for(asset,width)
